pipeline {
    agent any

    environment {
        DOCKERHUB_CREDENTIALS = 'dockerhub-creds'
        FRONTEND_IMAGE = 'rasoolp-ai/react-frontend'
        BACKEND_IMAGE = 'rasoolp-ai/go-backend'
        FRONTEND_TAG = 'latest'
        BACKEND_TAG = 'latest'
        REACT_APP_SERVER_URL = 'http://backend-service:8080/employees'  // internal DNS in Kubernetes
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build Frontend Image') {
            steps {
                dir('frontend') {
                    script {
                        docker.build("${FRONTEND_IMAGE}:${FRONTEND_TAG}", 
                          "--build-arg REACT_APP_SERVER_URL=${REACT_APP_SERVER_URL} .")
                    }
                }
            }
        }

        stage('Build Backend Image') {
            steps {
                dir('backend') {
                    script {
                        docker.build("${BACKEND_IMAGE}:${BACKEND_TAG}")
                    }
                }
            }
        }

        stage('Push Images to Docker Hub') {
            steps {
                withCredentials([usernamePassword(
                    credentialsId: "${DOCKERHUB_CREDENTIALS}",
                    usernameVariable: 'DOCKER_USER',
                    passwordVariable: 'DOCKER_PASS'
                )]) {
                    script {
                        docker.withRegistry('https://index.docker.io/v1/', "${DOCKERHUB_CREDENTIALS}") {
                            docker.image("${FRONTEND_IMAGE}:${FRONTEND_TAG}").push()
                            docker.image("${BACKEND_IMAGE}:${BACKEND_TAG}").push()
                        }
                    }
                }
            }
        }
    }

    post {
        success {
            echo "✅ Successfully built and pushed both frontend and backend images!"
        }
        failure {
            echo "❌ Something went wrong during the pipeline."
        }
    }
}
